void setBotSkills(int64_t botId,
                  const std::vector<std::pair<std::string, bool>>& skills) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  std::string ts = nowIso();
  execute("DELETE FROM bot_skills WHERE bot_id=?", {std::to_string(botId)});
  for (auto& [sid, on] : skills) {
    execute("INSERT INTO bot_skills(bot_id,skill_id,enabled,params,created_at,updated_at) "
            "VALUES(?,?,?,?,?,?)",
            {std::to_string(botId), sid, on ? "1" : "0", "{}", ts, ts});
  }
}

std::vector<BotSkill> listBotSkills(int64_t botId) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  std::vector<BotSkill> out;
  query("SELECT id,bot_id,skill_id,enabled,params,created_at,updated_at "
        "FROM bot_skills WHERE bot_id=? ORDER BY id ASC",
        {std::to_string(botId)},
        [&](sqlite3_stmt* st) { out.push_back(rowToSkill(st)); });
  return out;
}

Reminder addReminder(int64_t botId, const std::string& content,
                     const std::string& dueAt) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  std::string ts = nowIso();
  execute("INSERT INTO reminders(bot_id,content,due_at,done,created_at) VALUES(?,?,?,0,?)",
          {std::to_string(botId), content, dueAt, ts});
  Reminder r;
  r.id = sqlite3_last_insert_rowid(g_db);
  r.bot_id = botId;
  r.content = content;
  r.due_at = dueAt;
  r.done = false;
  r.created_at = ts;
  return r;
}

std::vector<Reminder> listReminders(bool dueOnly) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  std::vector<Reminder> out;
  std::string sql = "SELECT id,bot_id,content,due_at,done,created_at FROM reminders";
  if (dueOnly) {
    sql += " WHERE done=0 AND due_at <= ? ORDER BY due_at ASC";
    query(sql.c_str(), {nowIso()},
          [&](sqlite3_stmt* st) { out.push_back(rowToReminder(st)); });
  } else {
    sql += " ORDER BY due_at ASC";
    query(sql.c_str(), {}, [&](sqlite3_stmt* st) { out.push_back(rowToReminder(st)); });
  }
  return out;
}

bool markReminderDone(int64_t id) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  execute("UPDATE reminders SET done=1 WHERE id=?", {std::to_string(id)});
  return sqlite3_changes(g_db) > 0;
}
```

（`due_at <= ?` 与 `nowIso()` 均为 ISO 8601 本地时间字符串，字典序可比；测试用例的 2000/2099 年份字符串正因此可比较。）

namespace sk {

// ---------- calculator：递归下降四则运算（无 eval） ----------
struct CalcParser {
  const std::string& s;
  size_t i = 0;

  explicit CalcParser(const std::string& str) : s(str) {}

  void skipWs() {
    while (i < s.size() && std::isspace(static_cast<unsigned char>(s[i]))) ++i;
  }
  double parseExpr() {
    double v = parseTerm();
    for (;;) {
      skipWs();
      if (i < s.size() && s[i] == '+') { ++i; v += parseTerm(); }
      else if (i < s.size() && s[i] == '-') { ++i; v -= parseTerm(); }
      else break;
    }
    return v;
  }
  double parseTerm() {
    double v = parseFactor();
    for (;;) {
      skipWs();
      if (i < s.size() && s[i] == '*') { ++i; v *= parseFactor(); }
      else if (i < s.size() && s[i] == '/') {
        ++i;
        double d = parseFactor();
        if (d == 0.0) throw std::runtime_error("除数为零");
        v /= d;
      } else break;
    }
    return v;
  }
  double parseFactor() {
    skipWs();
    if (i >= s.size()) throw std::runtime_error("表达式不完整");
    if (s[i] == '(') {
      ++i;
      double v = parseExpr();
      skipWs();
      if (i >= s.size() || s[i] != ')') throw std::runtime_error("括号不匹配");
      ++i;
      return v;
    }
    if (s[i] == '-') { ++i; return -parseFactor(); }
    if (!std::isdigit(static_cast<unsigned char>(s[i])) && s[i] != '.')
      throw std::runtime_error("非法字符");
    size_t start = i;
    while (i < s.size() &&
           (std::isdigit(static_cast<unsigned char>(s[i])) || s[i] == '.'))
      ++i;
    try {
      return std::stod(s.substr(start, i - start));
    } catch (...) {
      throw std::runtime_error("数字解析失败");
    }
  }
};

json execCalculator(const json& params) {
  std::string expr = params.value("expression", "");
  if (expr.empty()) return {{"error", "表达式为空"}};
  try {
    CalcParser p(expr);
    double v = p.parseExpr();
    p.skipWs();
    if (p.i != p.s.size()) throw std::runtime_error("存在多余字符");
    return {{"result", v}};
  } catch (const std::exception& e) {
    return {{"error", std::string("计算失败：") + e.what()}};
  }
}

json execDatetime(const json&) {
  std::time_t t = std::time(nullptr);
  std::tm local{};
  localtime_s(&local, &t);
  char buf[64];
  std::strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &local);
  char date[32], weekday[32];
  std::strftime(date, sizeof(date), "%Y-%m-%d", &local);
  std::strftime(weekday, sizeof(weekday), "%A", &local);
  return {{"now", buf}, {"date", date}, {"weekday", weekday}};
}

json execSysinfo(const json&) {
  MEMORYSTATUSEX ms;
  ms.dwLength = sizeof(ms);
  if (!GlobalMemoryStatusEx(&ms)) return {{"error", "内存信息获取失败"}};
  ULARGE_INTEGER freeAvail, totalBytes, freeBytes;
  if (!GetDiskFreeSpaceExW(L"C:\\", &freeAvail, &totalBytes, &freeBytes))
    return {{"error", "磁盘信息获取失败"}};
  return {{"memory_total_mb", ms.ullTotalPhys / 1024 / 1024},
          {"memory_free_mb", ms.ullAvailPhys / 1024 / 1024},
          {"disk_free_gb", freeBytes.QuadPart / 1024 / 1024 / 1024}};
}

// 拒绝 localhost 与内网段（SSRF 防护）
bool isBlockedHost(const std::string& host) {
  std::string h = host;
  if (h == "localhost") return true;
  if (h.rfind("127.", 0) == 0) return true;
  if (h.rfind("10.", 0) == 0) return true;
  if (h.rfind("192.168.", 0) == 0) return true;
  if (h.rfind("172.", 0) == 0) {
    auto dot = h.find('.', 4);
    if (dot != std::string::npos) {
      int part = std::atoi(h.substr(4, dot - 4).c_str());
      if (part >= 16 && part <= 31) return true;
    }
  }
  return false;
}

std::wstring widen(const std::string& s) { return std::wstring(s.begin(), s.end()); }

json execHttpGet(const json& params) {
  std::string url = params.value("url", "");
  if (url.empty()) return {{"error", "url 为空"}};
  bool https = true;
  if (url.rfind("https://", 0) == 0) url = url.substr(8);
  else if (url.rfind("http://", 0) == 0) { url = url.substr(7); https = false; }
  else return {{"error", "仅支持 http/https"}};
  auto slash = url.find('/');
  std::string authority = slash == std::string::npos ? url : url.substr(0, slash);
  std::string path = slash == std::string::npos ? "/" : url.substr(slash);
  std::string host = authority;
  int port = https ? 443 : 80;
  auto colon = authority.find(':');
  if (colon != std::string::npos) {
    host = authority.substr(0, colon);
    try { port = std::stoi(authority.substr(colon + 1)); } catch (...) {}
  }
  if (isBlockedHost(host)) return {{"error", "拒绝访问本地/内网地址"}};

  HINTERNET hSess = WinHttpOpen(L"OruaSkill/1.0", WINHTTP_ACCESS_TYPE_NO_PROXY,
                                WINHTTP_NO_PROXY_NAME, WINHTTP_NO_PROXY_BYPASS, 0);
  if (!hSess) return {{"error", "WinHttpOpen 失败"}};
  WinHttpSetTimeouts(hSess, 3000, 3000, 10000, 10000);
  HINTERNET hConn = WinHttpConnect(hSess, widen(host).c_str(), port, 0);
  HINTERNET hReq = hConn ? WinHttpOpenRequest(hConn, L"GET", widen(path).c_str(),
                                              nullptr, WINHTTP_NO_REFERER,
                                              WINHTTP_DEFAULT_ACCEPT_TYPES,
                                              https ? WINHTTP_FLAG_SECURE : 0)
                         : nullptr;
  if (!hConn || !hReq) {
    if (hReq) WinHttpCloseHandle(hReq);
    if (hConn) WinHttpCloseHandle(hConn);
    WinHttpCloseHandle(hSess);
    return {{"error", "连接创建失败"}};
  }
  json out = {{"error", "请求失败"}};
  if (WinHttpSendRequest(hReq, nullptr, 0, nullptr, 0, 0, 0) &&
      WinHttpReceiveResponse(hReq, nullptr)) {
    std::string body;
    char buf[2048];
    DWORD read = 0, avail = 0;
    while (body.size() < 2048 && WinHttpQueryDataAvailable(hReq, &avail) && avail > 0 &&
           WinHttpReadData(hReq, buf, (std::min)(avail, (DWORD)sizeof(buf)), &read) &&
           read > 0)
      body.append(buf, read);
    if (body.size() > 2048) body.resize(2048);
    out = {{"body", body}};
  }
  WinHttpCloseHandle(hReq);
  WinHttpCloseHandle(hConn);
  WinHttpCloseHandle(hSess);
  return out;
}

json execFileRead(const json& params) {
  std::string p = params.value("path", "");
  if (p.empty()) return {{"error", "path 为空"}};
  std::ifstream ifs(p, std::ios::binary);
  if (!ifs) return {{"error", "文件打开失败"}};
  std::string content;
  char buf[4096];
  while (content.size() < 8192 && ifs.read(buf, sizeof(buf)))
    content.append(buf, ifs.gcount());
  if (ifs.gcount() > 0 && content.size() < 8192)
    content.append(buf, (std::min)(ifs.gcount(), (std::streamsize)(8192 - content.size())));
  return {{"content", content}};
}

void ToolCallAccumulator::feed(const json& delta) {
  if (!delta.contains("tool_calls") || !delta["tool_calls"].is_array()) return;
  for (auto& tc : delta["tool_calls"]) {
    int idx = tc.value("index", 0);
    while (static_cast<int>(out_.size()) <= idx) out_.push_back(json::object());
    auto& slot = out_[idx];
    if (tc.contains("id") && tc["id"].is_string()) slot["id"] = tc["id"];
    if (tc.value("type", "") == "function") slot["type"] = "function";
    if (tc.contains("function") && tc["function"].is_object()) {
      auto& fn = tc["function"];
      if (fn.contains("name") && fn["name"].is_string() && !fn["name"].get<std::string>().empty())
        slot["function"]["name"] = fn["name"];
      if (fn.contains("arguments") && fn["arguments"].is_string()) {
        std::string prev = slot["function"].value("arguments", "");
        slot["function"]["arguments"] = prev + fn["arguments"].get<std::string>();
      }
    }
  }
}

void registerSkillRoutes(httplib::Server& svr) {
  svr.Get("/api/v1/skills", [](const httplib::Request&, httplib::Response& res) {
    json arr = json::array();
    for (auto& s : sk::registry())

  });

  svr.Get(R"(/api/v1/bots/(\d+)/skills)", [](const httplib::Request& req, httplib::Response& res) {
    int64_t id = std::stoll(req.matches[1].str());
    if (!db::getBot(id)) {
      fail(res, 404, 404, "Bot 不存在");
      return;
    }
    json arr = json::array();
    for (auto& s : db::listBotSkills(id))
      arr.push_back({{"skill_id", s.skill_id}, {"enabled", s.enabled}});
    res.set_content(okJson(json{{"skills", arr}}).dump(), "application/json; charset=utf-8");
  });

    if (!db::getBot(id)) {
      fail(res, 404, 404, "Bot 不存在");
      return;
    }
    json b = parseBody(req, res);
    if (res.status == 400) return;
    if (!b.contains("skills") || !b["skills"].is_array()) {
      fail(res, 400, 400, "skills 必须为数组");
      return;
    }
    std::vector<std::pair<std::string, bool>> skills;
    for (auto& s : b["skills"]) {
      std::string sid = s.value("skill_id", "");
      bool on = s.value("enabled", true);
      bool known = false;
      for (auto& def : sk::registry())
        if (def.id == sid) known = true;
      if (!known) {
        fail(res, 400, 400, "未知技能：" + sid);
        return;
      }
      skills.push_back({sid, on});
    }
    db::setBotSkills(id, skills);
    res.set_content(okJson(json::object()).dump(), "application/json; charset=utf-8");
  });

using System.IO;
using System.Text.Json;
using System.Windows;
using Microsoft.Web.WebView2.Core;

namespace AiCompanion.App;

/// <summary>独立聊天窗口：复用同一前端（detached 模式），不持有后端守护。</summary>
public partial class ChatWindow : Window
{
    private readonly string _target;

    public ChatWindow(string target, int apiPort)
    {
        InitializeComponent();
        _target = target;
        Title = target.StartsWith("group:") ? "群聊" : "聊天";
        _apiPort = apiPort;
    }

    private readonly int _apiPort;

    private static string WwwrootDir => Path.Combine(AppContext.BaseDirectory, "wwwroot");

    private async void OnLoaded(object sender, RoutedEventArgs e)
    {
        await InitWebViewAsync();
    }

    private async Task InitWebViewAsync()
    {
        var env = await CoreWebView2Environment.CreateAsync(
            null, Path.Combine(AppContext.BaseDirectory, "WebView2Data-detached"));
        await WebView.EnsureCoreWebView2Async(env);
        WebView.CoreWebView2.SetVirtualHostNameToFolderMapping(
            "app.local", WwwrootDir, CoreWebView2HostResourceAccessKind.Allow);
        await WebView.CoreWebView2.AddScriptToExecuteOnDocumentCreatedAsync(
            $"window.__API_BASE__ = 'http://127.0.0.1:{_apiPort}';" +
            $"window.__DETACH_TARGET__ = '{_target}';");
        WebView.CoreWebView2.Navigate(
            "https://app.local/index.html?window=detached&target=" +
            System.Uri.EscapeDataString(_target));
    }
}
```

/* ================= 插件 ================= */
async function loadSkills() {
  const res = await App.api.get('/api/v1/skills');
  App.state.skills = res.skills || [];

}
function renderSkillCards() {
  const el = $('skills-list');
  el.innerHTML = '';
  for (const s of App.state.skills) {
    const card = document.createElement('div');
    card.className = 'skill-card';
    card.innerHTML = `<h4>${escapeHtml(s.name)}</h4><p>${escapeHtml(s.description)}</p>`;
    el.appendChild(card);
  }
}
async function loadBotSkills(botId) {
  const res = await App.api.get(`/api/v1/bots/${botId}/skills`);
  App.state.botSkills[botId] = res.skills || [];
  renderBotSkills();
}
function renderBotSkills() {
  const sel = $('skill-bot-select');
  sel.innerHTML = '';
  for (const b of App.state.bots) {
    const o = document.createElement('option');
    o.value = b.id;
    o.textContent = b.name;
    if (b.id === App.state.pluginBotId) o.selected = true;
    sel.appendChild(o);
  }
  const id = App.state.pluginBotId;
  const el = $('bot-skills-list');
  el.innerHTML = '';
  if (!id) return;
  const enabled = new Map((App.state.botSkills[id] || []).map((s) => [s.skill_id, s.enabled]));
  for (const s of App.state.skills) {
    const row = document.createElement('label');
    row.className = 'bot-skill-row';
    const on = enabled.get(s.id) ?? false;
    row.innerHTML = `<span>${escapeHtml(s.name)}</span>
      <input type="checkbox" data-skill="${escapeHtml(s.id)}" ${on ? 'checked' : ''}>`;
    row.querySelector('input').addEventListener('change', async (e) => {
      const skills = [];
      for (const sk of App.state.skills) {
        const c = e.target.parentElement.parentElement.querySelector(`[data-skill="${CSS.escape(sk.id)}"]`);
        if (c && c.checked) skills.push({ skill_id: sk.id, enabled: true });
      }
      await App.api.put(`/api/v1/bots/${id}/skills`, { skills });
      toast('技能已保存');
    });
    el.appendChild(row);
  }
}
