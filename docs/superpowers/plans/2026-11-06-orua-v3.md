> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 Orua 增加 Bot 长期记忆系统（提取/检索/注入/记忆库 UI），全 UI 术语「机器人」→「Bot」，Bot 编辑器对齐参考图字段并支持图片头像。

**Tech Stack:** C++20 / cpp-httplib / SQLite（沿用 v2）；前端原生 JS（WebView2）；测试 doctest + PowerShell 集成脚本；LLM_MOCK=1 全链路可测。

```cpp
Memory addMemory(int64_t botId, const std::string& scope, int64_t groupId,
                 const std::string& content, const std::string& tags,
                 int64_t sourceMsgId) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  std::string ts = nowIso();
  execute("INSERT INTO bot_memories(bot_id,scope,group_id,content,tags,source_msg_id,"
          "created_at,updated_at) VALUES(?,?,?,?,?,?,?,?)",
          {std::to_string(botId), scope, groupId == 0 ? "" : std::to_string(groupId),
           content, tags, sourceMsgId == 0 ? "" : std::to_string(sourceMsgId), ts, ts});
  Memory m;
  m.id = sqlite3_last_insert_rowid(g_db);
  m.bot_id = botId; m.scope = scope; m.group_id = groupId;
  m.content = content; m.tags = tags; m.source_msg_id = sourceMsgId;
  m.created_at = m.updated_at = ts;
  return m;
}

bool updateMemoryContent(int64_t id, const std::string& content) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  execute("UPDATE bot_memories SET content=?, updated_at=? WHERE id=?",
          {content, nowIso(), std::to_string(id)});
  return sqlite3_changes(g_db) > 0;
}

bool deleteMemory(int64_t id) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  execute("DELETE FROM bot_memories WHERE id=?", {std::to_string(id)});
  return sqlite3_changes(g_db) > 0;
}

std::vector<Memory> listMemories(int64_t botId, const std::string& scope,
                                 int64_t groupId, const std::string& q) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  std::vector<Memory> out;
  std::string sql =
      "SELECT id,bot_id,scope,COALESCE(group_id,0),content,tags,"
      "COALESCE(source_msg_id,0),created_at,updated_at "
      "FROM bot_memories WHERE bot_id=?";
  std::vector<std::string> params = {std::to_string(botId)};
  if (scope == "private" || scope == "group") {
    sql += " AND scope=?";
    params.push_back(scope);
  }
  if (groupId >= 0) {
    sql += " AND group_id=?";
    params.push_back(std::to_string(groupId));
  }
  if (!q.empty()) {
    sql += " AND (content LIKE ? OR tags LIKE ?)";
    params.push_back("%" + q + "%");
    params.push_back("%" + q + "%");
  }
  sql += " ORDER BY updated_at DESC, id DESC";
  query(sql.c_str(), params, [&](sqlite3_stmt* st) { out.push_back(rowToMemory(st)); });
  return out;
}

void deleteMemoriesByBot(int64_t botId) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  execute("DELETE FROM bot_memories WHERE bot_id=?", {std::to_string(botId)});
}

void deleteMemoriesByGroup(int64_t groupId) {
  std::lock_guard<std::recursive_mutex> lock(g_mtx);
  execute("DELETE FROM bot_memories WHERE group_id=?", {std::to_string(groupId)});
}

std::string dataDir() { return g_dataDir; }
```

```cpp
#include <deque>

#include "db.h"

namespace mem {

struct ExtractJob {
  int64_t bot_id = 0;
  std::string scope;   // "private" | "group"
  int64_t group_id = 0;
  std::string dialogue;

// 提取：异步、worker 串行、失败重试一次后放弃；LLM_MOCK 固定两条占位记忆。
// 检索：两段式（本地关键词召回 ≤30 → LLM 精选 ≤maxK），失败降级、无候选为空。
class MemoryService {
 public:
  static MemoryService& instance();

  void enqueueExtract(int64_t botId, const std::string& scope, int64_t groupId,
                      const std::string& dialogue);

                                       int64_t groupId, const std::string& dialogue,
                                       int maxK);

  MemoryService() = default;

  void ensureWorker();
  void workerLoop();
  void extractOnce(const ExtractJob& job);
  llm::ChatRequest buildExtractRequest(const db::Bot& bot);   // 与引擎同款解析
  void applyOps(const std::string& rawJson, const ExtractJob& job);

                                   const std::string& dialogue, int maxK);

  std::mutex mtx_;
  std::condition_variable cv_;
  std::deque<ExtractJob> queue_;
  std::thread worker_;
```

```cpp
#include "memory_service.h"

#include <algorithm>
#include <cctype>
#include <cstdlib>
#include <chrono>
#include <stdexcept>

#include "dpapi_util.h"
#include "json_util.h"
#include "llm_provider.h"
#include "logger.h"
#include "openai_provider.h"

namespace mem {
namespace {

std::vector<std::string> tokensOf(const std::string& text) {
  std::vector<std::string> out;
  size_t i = 0;
  std::string word;
  auto flush = [&] {
    if (word.size() >= 2) out.push_back(word);
    word.clear();
  };
  while (i < text.size()) {
    unsigned char c = static_cast<unsigned char>(text[i]);
    if (c < 0x80) {
      if (std::isalnum(c)) word += static_cast<char>(c);
      else flush();
      ++i;
    } else {
      flush();
      size_t len = (c >= 0xF0) ? 4 : (c >= 0xE0) ? 3 : 2;   // UTF-8 序列长度
      if (i + len <= text.size()) out.push_back(text.substr(i, len));   // 单个 CJK 字符
      i += len;
    }
  }
  flush();
  return out;
}

}  // namespace

MemoryService& MemoryService::instance() {
  static MemoryService svc;
  return svc;
}

void MemoryService::ensureWorker() {
  if (workerStarted_) return;
  workerStarted_ = true;
  worker_ = std::thread([this] { workerLoop(); });
}

void MemoryService::enqueueExtract(int64_t botId, const std::string& scope,
                                   int64_t groupId, const std::string& dialogue) {
  if (db::getSetting("memory_extraction").value_or("1") != "1") return;
  {
    std::lock_guard<std::mutex> lk(mtx_);
    queue_.push_back({botId, scope, groupId, dialogue});
    ensureWorker();
  }
  cv_.notify_one();
}

void MemoryService::workerLoop() {
  for (;;) {
    ExtractJob job;
    {
      std::unique_lock<std::mutex> lk(mtx_);
      cv_.wait(lk, [&] { return !queue_.empty(); });
      job = queue_.front();
      queue_.pop_front();
    }
    try {
      extractOnce(job);
    } catch (const std::exception& e) {
      logx::error(std::string("memory extract failed: ") + e.what());
      std::this_thread::sleep_for(std::chrono::seconds(2));
      try {
        extractOnce(job);
      } catch (const std::exception& e2) {
        logx::error(std::string("memory extract retry failed: ") + e2.what());
      }
    }
  }
}

llm::ChatRequest MemoryService::buildExtractRequest(const db::Bot& bot) {
  llm::ChatRequest req;
  req.base_url = !bot.api_base_url.empty() ? bot.api_base_url
                                           : db::getSetting("base_url").value_or("");
  std::string keyEnc = !bot.api_key_enc.empty() ? bot.api_key_enc
                                                : db::getSetting("api_key_enc").value_or("");
  if (!keyEnc.empty()) {
    try {
      req.api_key = dpapi::unprotect(keyEnc);
    } catch (...) {
    }
  }
  req.model = !bot.api_model.empty() ? bot.api_model : db::getSetting("model").value_or("");
  req.proxy_enabled = db::getSetting("proxy_enabled").value_or("0") == "1";
  req.proxy_host = db::getSetting("proxy_host").value_or("");
  try {
    req.proxy_port = std::stoi(db::getSetting("proxy_port").value_or("0"));
  } catch (...) {
    req.proxy_port = 0;
  }
  return req;
}

void MemoryService::extractOnce(const ExtractJob& job) {
  if (db::getSetting("memory_extraction").value_or("1") != "1") return;

  if (std::getenv("LLM_MOCK")) {
    // 固定两条：一条 new、一条 update 占位（规格 §4.1，测试确定性；幂等：重复提取不新增）
    auto existing = db::listMemories(job.bot_id, job.scope,
                                     job.scope == "group" ? job.group_id : -1);
    bool hasCoffee = false;
    for (auto& m : existing)
      if (m.content.find("咖啡") != std::string::npos) hasCoffee = true;
    if (hasCoffee && !existing.empty()) {
      db::updateMemoryContent(existing.front().id, existing.front().content);  // update 占位
    } else {
      db::addMemory(job.bot_id, job.scope, job.group_id, "记住：用户喜欢咖啡",
                    "[\"咖啡\",\"偏好\"]", 0);
    }
    if (existing.size() < 2)
      db::addMemory(job.bot_id, job.scope, job.group_id, "记住：助手应当先问好再回答",
                    "[\"礼仪\"]", 0);
    return;
  }

  auto bot = db::getBot(job.bot_id);
  if (!bot) return;

  llm::ChatRequest req = buildExtractRequest(*bot);
  std::string existingList;
  auto mems = db::listMemories(job.bot_id, job.scope,
                               job.scope == "group" ? job.group_id : -1);
  int shown = 0;
  for (auto& m : mems) {
    if (shown++ >= 200) break;   // 输入上限 200 条
    existingList += std::to_string(m.id) + ": " + m.content + "\n";
  }
  if (existingList.empty()) existingList = "（无）";

  req.messages.push_back(
      {"system",
       "你是记忆提取器。根据对话提炼值得长期记住的事实（用户偏好、名字、约定、重要信息）。"
       "只输出 JSON：{\"ops\":[{\"action\":\"new\",\"content\":\"...\",\"tags\":[\"...\"]},"
       "{\"action\":\"update\",\"memory_id\":3,\"content\":\"...\",\"tags\":[\"...\"]}]}"
       "没有值得记住的则输出 {\"ops\":[]}"});
  req.messages.push_back({"user", "现有记忆（id: 内容）：\n" + existingList +
                                      "\n对话：\n" + job.dialogue});

  std::string full;
  llm::OpenAiCompatProvider provider;
  std::atomic_bool stop{false};
  provider.streamChat(
      req, [&](const llm::StreamEvent& e) {
        if (e.type == "delta") full += e.content;
      },
      stop);
  applyOps(full, job);
}

void MemoryService::applyOps(const std::string& rawJson, const ExtractJob& job) {
  auto a = rawJson.find('{');
  auto b = rawJson.rfind('}');
  if (a == std::string::npos || b == std::string::npos || b < a) return;
  json j;
  try {
    j = json::parse(rawJson.substr(a, b - a + 1));
  } catch (...) {
    return;
  }
  if (!j.contains("ops") || !j["ops"].is_array()) return;
  for (auto& op : j["ops"]) {
    std::string action = op.value("action", "");
    std::string content = op.value("content", "");
    if (content.empty()) continue;
    std::string tags = "[]";
    if (op.contains("tags") && op["tags"].is_array()) tags = op["tags"].dump();
    if (action == "new") {
      db::addMemory(job.bot_id, job.scope, job.group_id, content, tags, 0);
    } else if (action == "update") {
      int64_t id = op.value("memory_id", (int64_t)0);
      if (id > 0) db::updateMemoryContent(id, content);
    }
  }
}

std::vector<db::Memory> MemoryService::recallLocal(int64_t botId,
                                                   const std::string& scope,
                                                   int64_t groupId,
                                                   const std::string& dialogue) {
  auto all = db::listMemories(botId, scope, scope == "group" ? groupId : -1);
  if (all.empty()) return {};
  auto toks = tokensOf(dialogue);
  if (toks.empty()) return {};
  struct Scored {
    db::Memory m;
    int score = 0;
    std::string upd;
  };
  std::vector<Scored> hits;
  for (auto& m : all) {
    int s = 0;
    for (auto& t : toks) {
      if (m.content.find(t) != std::string::npos) s += 2;
      if (m.tags.find(t) != std::string::npos) s += 1;
    }
    if (s > 0) hits.push_back({m, s, m.updated_at});
  }
  std::sort(hits.begin(), hits.end(), [](const Scored& x, const Scored& y) {
    if (x.score != y.score) return x.score > y.score;
    return x.upd > y.upd;
  });
  std::vector<db::Memory> out;
  for (size_t i = 0; i < hits.size() && i < 30; ++i) out.push_back(hits[i].m);
  return out;
}

std::vector<int64_t> MemoryService::selectByLlm(
    const std::vector<db::Memory>& candidates, const std::string& dialogue,
    int maxK) {
  if (candidates.empty()) return {};
  auto bot = db::getBot(candidates.front().bot_id);
  if (!bot) return {};
  llm::ChatRequest req = buildExtractRequest(*bot);
  std::string list;
  for (auto& c : candidates)
    list += std::to_string(c.id) + ": " + c.content + "\n";
  req.messages.push_back(
      {"system",
       "从候选记忆中选出与对话最相关的至多 " + std::to_string(maxK) +
           " 条。只输出 JSON 数组：[3, 7, ...]（无相关则 []）"});
  req.messages.push_back({"user", "候选记忆：\n" + list + "\n对话：\n" + dialogue});

  std::string full;
  llm::OpenAiCompatProvider provider;
  std::atomic_bool stop{false};
  try {
    provider.streamChat(
        req, [&](const llm::StreamEvent& e) {
          if (e.type == "delta") full += e.content;
        },
        stop);
    auto a = full.find('[');
    auto b = full.rfind(']');
    if (a == std::string::npos || b == std::string::npos || b < a) return {};
    json j = json::parse(full.substr(a, b - a + 1));
    std::vector<int64_t> ids;
    if (!j.is_array()) return {};
    for (auto& v : j) {
      if (!v.is_number()) continue;
      ids.push_back(v.get<int64_t>());
      if ((int)ids.size() >= maxK) break;
    }
    return ids;
  } catch (...) {
    return {};   // 失败 → 调用方降级
  }
}

std::vector<db::Memory> MemoryService::pickMemories(int64_t botId,
                                                    const std::string& scope,
                                                    int64_t groupId,
                                                    const std::string& dialogue,
                                                    int maxK) {
  auto cand = recallLocal(botId, scope, groupId, dialogue);
  if (cand.empty()) return {};
  std::vector<int64_t> ids;
  if (std::getenv("LLM_MOCK")) {
    for (size_t i = 0; i < cand.size() && i < (size_t)maxK; ++i)
      ids.push_back(cand[i].id);
  } else {
    ids = selectByLlm(cand, dialogue, maxK);
    if (ids.empty()) {
      // 降级：阶段 2 失败 → 阶段 1 前 maxK
      for (size_t i = 0; i < cand.size() && i < (size_t)maxK; ++i)
        ids.push_back(cand[i].id);
    }
  }
  std::vector<db::Memory> out;
  for (auto id : ids)
    for (auto& c : cand)
      if (c.id == id) {
        out.push_back(c);
        break;
      }
  return out;
}

}  // namespace mem
```

```cpp
// 标准 base64 解码（头像上传用），返回空串表示解码失败
std::string b64Decode(const std::string& in) {
  static const std::string tbl =
      "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
  std::string out;
  int val = 0, bits = -8;
  for (unsigned char c : in) {
    if (std::isspace(c) || c == '=') continue;
    size_t pos = tbl.find(static_cast<char>(c));
    if (pos == std::string::npos) return "";
    val = (val << 6) | static_cast<int>(pos);
    bits += 6;
    if (bits >= 0) {
      out.push_back(static_cast<char>((val >> bits) & 0xFF));
      bits -= 8;
    }
  }
  return out;
}

json memoryToJson(const db::Memory& m) {

  if (m.scope == "private" && m.source_msg_id > 0)
    if (auto s = db::memorySourceSession(m.source_msg_id)) sid = *s;
```

```js
/* ================= 记忆库 ================= */
function renderMemoryBots() {
  const el = $('memory-bot-list');
  el.innerHTML = '';
  const bots = App.state.bots || [];
  if (!App.state.memoryBotId && bots.length) App.state.memoryBotId = bots[0].id;
  for (const b of bots) {
    const item = document.createElement('div');
    item.className = 'list-item' + (b.id === App.state.memoryBotId ? ' active' : '');
    item.innerHTML = `<div class="list-item-avatar">${botAvatarHtml(b)}</div>
      <div class="list-item-body"><div class="list-item-title">${escapeHtml(b.name)}</div></div>`;
    item.onclick = () => { App.state.memoryBotId = b.id; renderMemoryBots(); loadMemories(); };
    el.appendChild(item);
  }
}

async function loadMemories(q = '') {
  const id = App.state.memoryBotId;
  const el = $('mem-list');
  el.innerHTML = '';
  if (!id) { $('mem-empty').classList.remove('hidden'); return; }
  const res = await App.api.get(`/api/v1/bots/${id}/memories${q ? `?q=${encodeURIComponent(q)}` : ''}`);
  App.state.memories = res.memories || [];
  $('mem-empty').classList.toggle('hidden', App.state.memories.length > 0);
  for (const m of App.state.memories) {
    const card = document.createElement('div');
    card.className = 'mem-item';
    const grp = App.state.groups.find((g) => g.id === m.group_id);
    const scopeLabel = m.scope === 'group' ? `群聊 · ${escapeHtml(grp?.name || '已删除群')}` : '私聊';
    const tags = (() => { try { return JSON.parse(m.tags || '[]'); } catch (_) { return []; } })();
    const srcLink = m.source_session_id
      ? `<button class="btn-ghost" data-src="${m.source_session_id}">来源</button>` : '';
    card.innerHTML = `
      <div class="mem-content">${escapeHtml(m.content)}</div>
      <div class="mem-meta">
        <span class="mem-scope">${scopeLabel}</span>
        ${tags.map((t) => `<span>#${escapeHtml(t)}</span>`).join(' ')}
        <span>${escapeHtml(m.updated_at || '')}</span>
        <span class="mem-actions">
          ${srcLink}
          <button class="btn-ghost" data-edit="${m.id}">编辑</button>
          <button class="btn-ghost" data-del="${m.id}">删除</button>
        </span>
      </div>`;
    card.querySelector('[data-edit]')?.addEventListener('click', () => openMemoryEditor(m));
    card.querySelector('[data-del]')?.addEventListener('click', async () => {
      if (!confirm('删除这条记忆？')) return;
      await App.api.del(`/api/v1/bots/${id}/memories/${m.id}`);
      toast('记忆已删除');
      loadMemories($('mem-search').value.trim());
    });
    card.querySelector('[data-src]')?.addEventListener('click', () => {
      App.run('select-session', Number(card.querySelector('[data-src]').dataset.src));
      switchView('messages');
    });
    el.appendChild(card);
  }
}

let editingMemoryId = null;
function openMemoryEditor(m) {
  editingMemoryId = m.id;
  $('me-content').value = m.content;
  $('modal-memory').classList.remove('hidden');
}
```
